```python
"""Testes unitários para naturalsize, escritos a partir das regras de negócio R1 a R9."""

import pytest

from alvo import naturalsize


DECIMAL_SUFFIXES = ["kB", "MB", "GB", "TB", "PB", "EB", "ZB", "YB", "RB", "QB"]
BINARY_SUFFIXES = ["KiB", "MiB", "GiB", "TiB", "PiB", "EiB", "ZiB", "YiB", "RiB", "QiB"]
GNU_SUFFIXES = ["K", "M", "G", "T", "P", "E", "Z", "Y", "R", "Q"]


# ---------------------------------------------------------------------------
# R1: conversão com float(); as exceções de float() são propagadas
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        ("3000000", "3.0 MB"),
        ("1000", "1.0 kB"),
        ("999", "999 Bytes"),
        ("0", "0 Bytes"),
        ("1", "1 Byte"),
        ("-1", "-1 Byte"),
        ("1e6", "1.0 MB"),
        ("  1000  ", "1.0 kB"),
        ("-3000000", "-3.0 MB"),
    ],
)
def test_r1_numeric_string_is_converted(value, expected):
    assert naturalsize(value) == expected


@pytest.mark.parametrize(
    "value, expected",
    [
        (3000000, "3.0 MB"),
        (3000000.0, "3.0 MB"),
        (1000.0, "1.0 kB"),
        (1500.5, "1.5 kB"),
        (0.0, "0 Bytes"),
        (1.0, "1 Byte"),
    ],
)
def test_r1_int_and_float_are_accepted(value, expected):
    assert naturalsize(value) == expected


def test_r1_numeric_string_in_binary_mode():
    assert naturalsize("4096", binary=True) == "4.0 KiB"


def test_r1_numeric_string_in_gnu_mode():
    assert naturalsize("3000", gnu=True) == "2.9K"


@pytest.mark.parametrize("value", ["abc", "", "10 kB", "1,5", "one", "   "])
def test_r1_non_numeric_string_raises_value_error(value):
    with pytest.raises(ValueError):
        naturalsize(value)


@pytest.mark.parametrize("value", ["abc", "10 kB"])
def test_r1_non_numeric_string_raises_value_error_in_binary_mode(value):
    with pytest.raises(ValueError):
        naturalsize(value, binary=True)


@pytest.mark.parametrize("value", ["abc", "10 kB"])
def test_r1_non_numeric_string_raises_value_error_in_gnu_mode(value):
    with pytest.raises(ValueError):
        naturalsize(value, gnu=True)


def test_r1_none_raises_type_error():
    with pytest.raises(TypeError):
        naturalsize(None)


def test_r1_none_raises_type_error_in_binary_and_gnu_modes():
    with pytest.raises(TypeError):
        naturalsize(None, binary=True)
    with pytest.raises(TypeError):
        naturalsize(None, gnu=True)


@pytest.mark.parametrize("value", [[], {}, object()])
def test_r1_non_convertible_types_raise_type_error(value):
    with pytest.raises(TypeError):
        naturalsize(value)


# ---------------------------------------------------------------------------
# R2: valor absoluto exatamente 1 fora do modo GNU retorna "1 Byte" (singular)
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (1, "1 Byte"),
        (-1, "-1 Byte"),
        (1.0, "1 Byte"),
        (-1.0, "-1 Byte"),
        ("1", "1 Byte"),
        ("-1", "-1 Byte"),
    ],
)
def test_r2_one_byte_singular_with_sign_preserved(value, expected):
    assert naturalsize(value) == expected


def test_r2_one_byte_in_binary_mode():
    assert naturalsize(1, binary=True) == "1 Byte"
    assert naturalsize(-1, binary=True) == "-1 Byte"


def test_r2_one_byte_ignores_format():
    assert naturalsize(1, format="%.3f") == "1 Byte"


@pytest.mark.parametrize("value, expected", [(2, "2 Bytes"), (-2, "-2 Bytes"), (0, "0 Bytes")])
def test_r2_boundary_neighbors_of_one_are_plural(value, expected):
    assert naturalsize(value) == expected


# ---------------------------------------------------------------------------
# R3: valor absoluto abaixo da base retorna a parte inteira + " Bytes" ("<n>B" no GNU)
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (0, "0 Bytes"),
        (2, "2 Bytes"),
        (500, "500 Bytes"),
        (998, "998 Bytes"),
        (999, "999 Bytes"),
        (-500, "-500 Bytes"),
        (-999, "-999 Bytes"),
    ],
)
def test_r3_decimal_below_base_returns_bytes(value, expected):
    assert naturalsize(value) == expected


@pytest.mark.parametrize(
    "value, expected",
    [
        (0, "0 Bytes"),
        (2, "2 Bytes"),
        (999, "999 Bytes"),
        (1000, "1000 Bytes"),
        (1023, "1023 Bytes"),
    ],
)
def test_r3_binary_below_base_1024_returns_bytes(value, expected):
    assert naturalsize(value, binary=True) == expected


@pytest.mark.parametrize(
    "value, expected",
    [
        (0, "0B"),
        (1, "1B"),
        (-1, "-1B"),
        (2, "2B"),
        (300, "300B"),
        (999, "999B"),
        (1000, "1000B"),
        (1023, "1023B"),
    ],
)
def test_r3_gnu_below_base_returns_n_b_without_space(value, expected):
    assert naturalsize(value, gnu=True) == expected


def test_r3_gnu_one_is_not_singular_byte():
    assert naturalsize(1, gnu=True) == "1B"
    assert naturalsize(1, binary=True, gnu=True) == "1B"


def test_r3_below_base_ignores_format():
    assert naturalsize(999, format="%.3f") == "999 Bytes"
    assert naturalsize(1023, binary=True, format="%.3f") == "1023 Bytes"
    assert naturalsize(300, gnu=True, format="%.3f") == "300B"


# ---------------------------------------------------------------------------
# R4: modo decimal (padrão), base 1000, sufixos kB..QB, separados por espaço
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (1000, "1.0 kB"),
        (1001, "1.0 kB"),
        (1500, "1.5 kB"),
        (3000000, "3.0 MB"),
        (123_456_789, "123.5 MB"),
        (2_500_000_000, "2.5 GB"),
        (10**28, "10.0 RB"),
    ],
)
def test_r4_decimal_mode_examples(value, expected):
    assert naturalsize(value) == expected


@pytest.mark.parametrize(
    "exponent, suffix",
    list(enumerate(DECIMAL_SUFFIXES, start=1)),
)
def test_r4_decimal_suffix_sequence(exponent, suffix):
    assert naturalsize(1000**exponent) == "1.0 " + suffix


@pytest.mark.parametrize(
    "value, expected",
    [
        (-1000, "-1.0 kB"),
        (-1500, "-1.5 kB"),
        (-3000000, "-3.0 MB"),
        (-2_500_000_000, "-2.5 GB"),
    ],
)
def test_r4_decimal_sign_is_preserved(value, expected):
    assert naturalsize(value) == expected


def test_r4_decimal_is_default_mode():
    assert naturalsize(3000000) == naturalsize(3000000, binary=False, gnu=False)
    assert naturalsize(3000000, False, False) == "3.0 MB"


def test_r4_decimal_lower_boundary_switches_from_bytes_to_kb():
    assert naturalsize(999) == "999 Bytes"
    assert naturalsize(1000) == "1.0 kB"


# ---------------------------------------------------------------------------
# R5: binary=True, base 1024, sufixos KiB..QiB
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (1024, "1.0 KiB"),
        (1025, "1.0 KiB"),
        (1536, "1.5 KiB"),
        (3000, "2.9 KiB"),
        (4096, "4.0 KiB"),
        (3 * 1024**3, "3.0 GiB"),
    ],
)
def test_r5_binary_mode_examples(value, expected):
    assert naturalsize(value, binary=True) == expected


def test_r5_binary_positional_argument():
    assert naturalsize(3000, True) == "2.9 KiB"


@pytest.mark.parametrize(
    "value, expected",
    [
        (-4096, "-4.0 KiB"),
        (-1024, "-1.0 KiB"),
        (-1536, "-1.5 KiB"),
    ],
)
def test_r5_binary_sign_is_preserved(value, expected):
    assert naturalsize(value, binary=True) == expected


@pytest.mark.parametrize(
    "exponent, suffix",
    list(enumerate(BINARY_SUFFIXES, start=1)),
)
def test_r5_binary_suffix_sequence(exponent, suffix):
    assert naturalsize(1024**exponent, binary=True) == "1.0 " + suffix


def test_r5_binary_lower_boundary_switches_from_bytes_to_kib():
    assert naturalsize(1023, binary=True) == "1023 Bytes"
    assert naturalsize(1024, binary=True) == "1.0 KiB"


# ---------------------------------------------------------------------------
# R6: gnu=True, base 1024, sufixos de uma letra sem espaço; binary é ignorado
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (1024, "1.0K"),
        (1536, "1.5K"),
        (3000, "2.9K"),
        (4096, "4.0K"),
        (3 * 1024**3, "3.0G"),
    ],
)
def test_r6_gnu_mode_examples(value, expected):
    assert naturalsize(value, gnu=True) == expected


def test_r6_gnu_positional_arguments():
    assert naturalsize(3000, False, True) == "2.9K"


@pytest.mark.parametrize(
    "exponent, suffix",
    list(enumerate(GNU_SUFFIXES, start=1)),
)
def test_r6_gnu_suffix_sequence_without_space(exponent, suffix):
    result = naturalsize(1024**exponent, gnu=True)
    assert result == "1.0" + suffix
    assert " " not in result


@pytest.mark.parametrize("value", [1023, 1024, 3000, 1024**3, 3 * 1024**5])
def test_r6_gnu_ignores_binary_parameter(value):
    assert naturalsize(value, binary=True, gnu=True) == naturalsize(value, binary=False, gnu=True)


def test_r6_gnu_with_binary_true_uses_one_letter_suffix():
    assert naturalsize(3000, True, True) == "2.9K"


def test_r6_gnu_uses_base_1024_not_1000():
    assert naturalsize(1000, gnu=True) == "1000B"
    assert naturalsize(1024, gnu=True) == "1.0K"


# ---------------------------------------------------------------------------
# R7: o parâmetro format define como o número é formatado
# ---------------------------------------------------------------------------

def test_r7_format_example_gnu():
    assert naturalsize(3000, False, True, "%.3f") == "2.930K"


@pytest.mark.parametrize(
    "value, fmt, expected",
    [
        (1234567, "%.3f", "1.235 MB"),
        (3000000, "%.2f", "3.00 MB"),
        (3000000, "%.0f", "3 MB"),
        (1500, "%.4f", "1.5000 kB"),
    ],
)
def test_r7_format_in_decimal_mode(value, fmt, expected):
    assert naturalsize(value, format=fmt) == expected


def test_r7_format_in_binary_mode():
    assert naturalsize(3000, binary=True, format="%.2f") == "2.93 KiB"
    assert naturalsize(3000, binary=True, format="%.3f") == "2.930 KiB"


def test_r7_format_keyword_in_gnu_mode():
    assert naturalsize(3000, gnu=True, format="%.3f") == "2.930K"


def test_r7_default_format_is_one_decimal_place():
    assert naturalsize(1234567) == "1.2 MB"
    assert naturalsize(1234567) == naturalsize(1234567, format="%.1f")


def test_r7_format_preserves_sign():
    assert naturalsize(-3000000, format="%.2f") == "-3.00 MB"


# ---------------------------------------------------------------------------
# R8: se o arredondamento levar o número a alcançar a base, usa-se o sufixo seguinte
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (999_999, "1.0 MB"),
        (-999_999, "-1.0 MB"),
        (999_999_999, "1.0 GB"),
        (999_999 * 10**24, "1.0 QB"),
    ],
)
def test_r8_decimal_rounding_to_base_promotes_suffix(value, expected):
    assert naturalsize(value) == expected


def test_r8_decimal_just_below_rounding_threshold_keeps_suffix():
    assert naturalsize(999_949) == "999.9 kB"


def test_r8_binary_rounding_to_base_promotes_suffix():
    assert naturalsize(1024**2 - 1, binary=True) == "1.0 MiB"
    assert naturalsize(1024**3 - 1, binary=True) == "1.0 GiB"


def test_r8_binary_just_below_rounding_threshold_keeps_suffix():
    assert naturalsize(1048000, binary=True) == "1023.4 KiB"


def test_r8_gnu_rounding_to_base_promotes_suffix():
    assert naturalsize(1024**2 - 1, gnu=True) == "1.0M"


def test_r8_gnu_just_below_rounding_threshold_keeps_suffix():
    assert naturalsize(1048000, gnu=True) == "1023.4K"


def test_r8_more_precise_format_does_not_reach_base():
    assert naturalsize(999_999, format="%.3f") == "999.999 kB"


# ---------------------------------------------------------------------------
# R9: acima do maior sufixo, o número continua crescendo com esse sufixo
# ---------------------------------------------------------------------------

@pytest.mark.parametrize(
    "value, expected",
    [
        (10**30, "1.0 QB"),
        (10**33, "1000.0 QB"),
        (10**34 * 3, "30000.0 QB"),
        (10**36, "1000000.0 QB"),
        (-(10**34) * 3, "-30000.0 QB"),
    ],
)
def test_r9_decimal_beyond_largest_suffix(value, expected):
    assert naturalsize(value) == expected


@pytest.mark.parametrize(
    "value, expected",
    [
        (1024**10, "1.0 QiB"),
        (1024**11, "1024.0 QiB"),
        (2**111, "2048.0 QiB"),
    ],
)
def test_r9_binary_beyond_largest_suffix(value, expected):
    assert naturalsize(value, binary=True) == expected


@pytest.mark.parametrize(
    "value, expected",
    [
        (1024**10, "1.0Q"),
        (1024**11, "1024.0Q"),
        (2**111, "2048.0Q"),
    ],
)
def test_r9_gnu_beyond_largest_suffix(value, expected):
    assert naturalsize(value, gnu=True) == expected
```
